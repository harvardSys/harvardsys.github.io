{{- /* Renders one image figure. Shared by the markdown image render hook (layouts/_default/_markup/render-image.html) and the {{< figure >}} shortcode (layouts/shortcodes/figure.html), so both produce identical markup. Context: a dict with page the page (for resolving page-bundle resources) required src image path or URL required alt alt text (defaults to the plain-text caption) caption caption, markdown allowed title optional heading shown above the caption (Hugo's built-in figure param) attr, attrlink attribution text / link appended to the caption size small | medium | large | full | wide (see assets/css/figures.css) panels "x,y,w,h; x,y,w,h; ..." crop boxes in the image's own units (pixels for raster images, viewBox units for SVG). Below 600px the figure shows these crops stacked instead of the whole image. class, id, width, height, loading, link, target, rel passed through */ -}} {{- $page := .page -}} {{- $src := .src -}} {{- read the intrinsic size. */ -}} {{- $w := 0 -}} {{- $h := 0 -}} {{- $vb := slice -}} {{- $u := urls.Parse $src -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or ($page.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- if eq .MediaType.SubType "svg" -}} {{- with index (findRE `]*>` .Content 1) 0 -}} {{- $tag := . -}} {{- $toPx := dict "" 1.0 "px" 1.0 "pt" 1.3333333 "pc" 16.0 "mm" 3.7795276 "cm" 37.795276 "in" 96.0 -}} {{- with findRESubmatch `\swidth="([0-9.]+)(px|pt|pc|mm|cm|in)?"` $tag 1 -}} {{- $m := index . 0 -}}{{- $w = mul (float (index $m 1)) (index $toPx (index $m 2)) -}} {{- end -}} {{- with findRESubmatch `\sheight="([0-9.]+)(px|pt|pc|mm|cm|in)?"` $tag 1 -}} {{- $m := index . 0 -}}{{- $h = mul (float (index $m 1)) (index $toPx (index $m 2)) -}} {{- end -}} {{- with findRESubmatch `viewBox="\s*([-0-9.]+)[\s,]+([-0-9.]+)[\s,]+([0-9.]+)[\s,]+([0-9.]+)` $tag 1 -}} {{- $m := index . 0 -}} {{- $vb = slice (float (index $m 1)) (float (index $m 2)) (float (index $m 3)) (float (index $m 4)) -}} {{- if or (eq $w 0) (eq $h 0) -}} {{- $w = index $vb 2 -}}{{- $h = index $vb 3 -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Panel crop boxes are given in the image's own units (viewBox units for SVG). */ -}} {{- $unitW := $w -}}{{- $unitH := $h -}} {{- with $vb -}}{{- $unitW = index . 2 -}}{{- $unitH = index . 3 -}}{{- end -}} {{- $panels := slice -}} {{- with .panels -}} {{- if not (and $unitW $unitH) -}} {{- errorf "%s: figure %q has panels but its size is unknown" $page.File.Path $src -}} {{- end -}} {{- range split . ";" -}} {{- $p := split (trim . " \n\t") "," -}} {{- if eq (len $p) 4 -}} {{- $panels = $panels | append (dict "x" (float (trim (index $p 0) " ")) "y" (float (trim (index $p 1) " ")) "w" (float (trim (index $p 2) " ")) "h" (float (trim (index $p 3) " "))) -}} {{- else if trim .
